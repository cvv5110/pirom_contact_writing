\paragraph*{Variational Formulation} Choosing appropriate basis functions $v(x)$ and $w(x)$, the variational formulation of the governing equations is obtained by applying the DG method element-wise to each component of the TPS. In contrast to the standard perfect-contact formulation, the component interfaces support a finite temperature jump across the interface caused by imperfect thermal contact. For each element, let $\cN_i$ denote the collection of elements neighboring element $E_i$, where the following separation of neighboring elements is introduced,
\begin{equation}
    \cN_i = \cN_i^{b}\cup\cN_{ir}^{c},\quad i=1,2,\dots,M, r=1,2,\dots,N_{\Gamma}
\end{equation}
Here, $\cN_i^{b}$ denotes the ordinary neighboring elements coupled by the standard DG interface terms through the interface $e_{ij}$, while $\cN_{ir}^{c}$ denotes the neighboring elements separated from $E_i$ by a an imperfect thermal contact interface $\Gamma_{ir}\subseteq\Gamma_r$, with $r\in\left\{1,2,\dots,N_{\Gamma}\right\}$ and $N_{\Gamma}$ is the total number of imperfect-contact interfaces in the TPS. Formally, the sets are defined as,
\begin{subequations}
    \begin{align}
        \cN_i^{b} &= \left\{j\in\{1,\ldots,M\} \:|\: E_j\cap E_i = e_{ij}\neq\emptyset, e_{ij}\subseteq\Omega\right\},\\
        \cN_{ir}^{c} &= \left\{j\in\{1,\ldots,M\} \:|\: \partial E_i\cap \partial E_j \cap\Gamma_r \neq \emptyset\right\}
    \end{align}
\end{subequations}
Under this separation, the Interior Penalty Galerkin (IPG) scheme yields the following bi-linear form for the variational formulation of the governing equations [Cohen2018],
\begin{equation}
    \sum_{i=1}^{M}a_{\epsilon,i}(v,w) = \sum_{i=1}^{M}\left( L_i^{(c)}(v) + L^{(b)}_i(v) \right)
\end{equation}
where $\epsilon$ is a user-specified parameter, and,
\begin{subequations}
    \begin{align}
        a_{\epsilon,i}(v,w) &= \int_{E_i}\left(\rho_i c_{p,i} v\frac{\partial w}{\partial t} + \vk_i\nabla v\cdot\nabla w\right)dE_i - \sum_{j\in\cN_i^{b}\cup\left\{T\right\}}\int_{e_{ij}}\left\{\vk_i\nabla v\cdot \vn_i\right\}\left[w\right]de_{ij}\notag\\
        &\quad + \epsilon \sum_{j\in\cN_i^{b}\cup\left\{T\right\}}\int_{e_{ij}}\left\{\vk_i\nabla w\cdot \vn_i\right\}\left[v\right]de_{ij} + \sigma\sum_{j\in\cN_i^{b}\cup\left\{T\right\}}\int_{e_{ij}}\left[v\right]\left[w\right]de_{ij}\\
        L^{(c)}_{i}(v) &= \sum_{r=1}^{N_{\Gamma}} \sum_{l\in\cN_{ir}^{c}}\int_{\Gamma_{il}} v q^{(c,r)}_{il}d\Gamma_{il}\\
        L^{(b)}_i(v) &= \epsilon\sum_{j\in\cN_i^{b}\cup\left\{T\right\}}\int_{e_{ij}} \left(\vk_i\nabla w\cdot \vn_i\right) T_b de_{ij} + \int_{e_{iq}} v q_b de_{iq} + \sigma\int_{e_{iT}} v T_b de_{iT}
    \end{align}
\end{subequations}
In the bi-linear form above, the notations of $\left[\right]$ and $\left\{\right\}$ are respectively the jumps and averages at a boundary $e_{ij}$ shared by two neighboring elements $E_i$ and $E_j$, defined as,
\begin{equation}
    \left[u\right] = u\Big|_{E_i} - u\Big|_{E_j},\quad \left\{u\right\} = \frac{1}{2}\left(u\Big|_{E_i} + u\Big|_{E_j}\right),\quad x\in e_{ij}=\partial E_i\cap\partial E_j
\end{equation}
Furthermore, the terms associated with $\sigma$ are introduced to enforce the Dirichlet boundary conditions, where $\sigma$ is a user-specified penalty parameter whose value depend on the size of an element. Depending on the choice of $\epsilon$, the bi-linear form may correspond to symmetric IPG ($\epsilon=-1$), non-symmetric IPG ($\epsilon=1$), or incomplete IPG ($\epsilon=0$) schemes. All these schemes are consistent with the original PDE and have similar convergence rate with respect to mesh size.

\paragraph*{Galerkin Projections} Next, the DG-based FOM is written in an element-wise form. For the $i$-th element, use a set of $P$ basis functions, such as polynomials, to represent the temperature field as in Eq. [X] that satisfy the conditions in Eq. [X]. Using test functions same as trial functions, the dynamics of the coefficients $\vu_i(t) = \left[u_{i,1}(t),\ldots,u_{i,P}(t)\right]^\top$ for the $i$-th element is obtained by evaluating the element-wise bi-linear forms,
\begin{equation}
    a_{\epsilon,i}(\phi_k^i,T) = L^{(c)}_i(\phi_k^i) + L^{(b)}_i(\phi_k^i),\quad k=1,\ldots,P
\end{equation}
where $\phi_k^i$ is the $k$-th basis function for the $i$-th element, where the conditions from Eq. [X] are satisfied. Evaluating the bi-linear forms leads to the following system of ODEs,
\begin{equation}
    \vA_i(\vu_i)\vu_i = \vB_i(\vu_i)\vu_i + \sum_{j\in\cN_i^{b}}\left(\vB_{ij}^{i}\vu_i + \vB_{ij}^{j}\vu_j  \right) + \vf_{i}^{(c)}(\vu,t) + \vf_i^{(b)}
\end{equation}
where, for $k,l=1,2,\dots,P$,
\begin{subequations}
    \begin{align}
        \left[\vA_i\right]_{kl} &= \int_{E_i}\rho_i c_{p,i}\phi_k^i\phi_l^i dE_i\\
        \left[\vB_i\right]_{kl} &= \int_{E_i}\vk_i\nabla\phi_k^i\cdot\nabla\phi_l^i dE_i\\
        \left[\vB_{ij}^{i}\right]_{kl} &= \left\{\begin{matrix}
            &-\int_{e_{ij}}\left\{\vk\nabla\phi_k^i\cdot\vn_i\right\}\phi_l^i + \sigma\left[\phi_k^i\right]\phi_l^id\Gamma, & j\in\cN_i^{b}\cup\left\{T\right\}\\
            &0, & \text{otherwise}
        \end{matrix}\right.\\
        \left[\vB_{ij}^{j}\right]_{kl} &= \left\{\begin{matrix}
            &\int_{e_{ij}}\left\{\vk\nabla\phi_k^i\cdot\vn\right\}\phi_l^j + \sigma\left[\phi_k^i\right]\phi_l^jd\Gamma, & j\in\cN_i^{b}\cup\left\{T\right\}\\
            &0, & \text{otherwise}
        \end{matrix}\right.\\
        \left[\vf_{i}^{(c)}\right]_{k} &= \left\{\begin{matrix}
            &\sum_{r=1}^{N_{\Gamma}} \sum_{l\in\cN_{ir}^{c}} \int_{\Gamma_{il}} \phi_k^i q^{(c,r)}_{il}d\Gamma, & l\in\cN_{ir}^{c}\\
            &0, & \text{otherwise}
        \end{matrix}\right.\\
        \left[\vf_i^{(b)}\right]_{k} &= \int_{e_{iq}}\phi_k^iq_b de_{iq} 
    \end{align}
\end{subequations}
Note that the matrices $\vA_i$, $\vB_i$, $\vB_{ij}^i$, and $\vB_{ij}^j$ depend on $\rho$, $c_p$, and $\vk$, respectively, and hence can be non-linear functions of the state $\vu_i$ if the material properties are temperature-dependent. Since the basis functions are orthogonal, if $\rho c_p$ is constant within an element, then $\vA_i$ is a diagonal matrix; otherwise, $\vA_i$ is symmetric and positive-definite, as $\rho c_p > 0$.

For compactness, the element-wise ODEs are stacked together to obtain the following system of ODEs for the entire TPS,
\begin{equation}
    \vA(\vu)\dot{\vu} = \vB(\vu)\vu + \vf^{(c)}(\vu,t) + \vf^{(b)}\label{eqn_compact_dg_fem}
\end{equation}
where $\vu = \left[\vu_1^\top,\ldots,\vu_M^\top\right]^\top\in\mathbb{R}^{MP}$ and $\vf^{(b)} = \left[\vf_1^{(b)\top},\ldots,\vf_M^{(b)\top}\right]^\top\in\mathbb{R}^{MP}$, and for $i,j=1,2,\dots,M$, it follows that,
\begin{subequations}
    \begin{align}
        \vB_{ij} &= \left\{
        \begin{matrix}
            &\vB_i(\vu) + \sum_{j\in\cN_i^{b}\cup\left\{T\right\}}\vB_{ij}^i, & i=j\\
            &\vB_{ij}^j, & i\neq j, j\in\cN_i^{b}\cup\left\{T\right\}\\
            &\vzero, &i\neq j, j\in\cN_i^c\\
            &\vzero, &\text{otherwise}
        \end{matrix}
        \right.\\
        \vf_i^{(c)} &= \left\{
            \begin{matrix}
                &\sum_{r=1}^{N_{\Gamma}} \sum_{l\in\cN_{ir}^c} \vf_{il}^{(c,r)}(\vu,t), & l\in\cN_{ir}^c\\
                &\vzero, & \text{otherwise}
            \end{matrix}
        \right.
    \end{align}
\end{subequations}
where $\cN_i^{c} = \bigcup_{r=1}^{N_{\Gamma}} \cN_{ir}^{c}$. The dependency of $\vA$, $\vB$, and $q_{il}^{(c,r)}$ on $\vu$ is explicitly noted in Eq. [X], which is the main source of non-linearities in the system.

Particularly, the contact heat flux $q^{(c,r)}_{il}$ between elements $E_i$ and $E_l$ across the imperfect-contact interface $\Gamma_r$ becomes a function of the DG-FEM state $\vu$ through the following relationship,

    \end{align}
\end{subequations}

Moreover, on the imperfect-contact interface $\Gamma_{l}$, the interface represents a physical thermal resistance between two neighboring components. Let $R_{x,l}$ denote the thermal contact resistance and define the corresponding thermal contact conductance as $h_{x,l} = 1/R_{x,l}$. Then, the heat flux across the interface is governed by,
\begin{equation}
    q^{(c)}_{ij} = h_{x,ij}\left(T^{-}_i(x,t) - T^{+}_j(x,t)\right), \quad x\in\Gamma_{ij}
\end{equation}
where $T^{-}_i$ and $T^{+}_j$ denote the temperature traces on the minus and plus sides of the interface $\Gamma_{ij}$, respectively. The normal conditions on the interface are given by the conservation of energy across the interface,
\begin{subequations}
    \begin{align}
        -\kappa_i\nabla T^{-}_i(x,t)\cdot\vn_i &= q^{(c)}_{ij}, \quad x\in\Gamma_{ij},\\
        -\kappa_j\nabla T^{+}_j(x,t)\cdot\vn_j &= -q^{(c)}_{ij}, \quad x\in\Gamma_{ij}
    \end{align}
\end{subequations}
where $\vn_i=-\vn_j$ is the unit normal vector of the interface $\Gamma_{ij}$ pointing from component $i$ to component $j$. Thus, across an imperfect contact interface, the heat flux is conserved, while the temperature is allowed to be discontinuous.

\paragraph*{Coarse-Graining Example} Consider the three-layered TPS shown in Fig. [X] with three components $\left\{\Omega_j\right\}_{j=1}^3$ and three contact interfaces $\left\{\Gamma_k\right\}_{k=1}^3$. The set of elements for each component, and on the minus and plus sides of each interface are defined as,
\begin{subequations}
    \begin{align}
        \cV_j &= \left\{m\in\{1,\ldots,M\}:E_m\subseteq\Omega_j\right\},\\
        \cF_k^{-} &= \left\{m\in\{1,\ldots,M\}:E_m\cap\Gamma_k\neq\emptyset, E_m\subseteq\Omega_{l(k)}\right\},\\
        \cF_k^{+} &= \left\{m\in\{1,\ldots,M\}:E_m\cap\Gamma_k\neq\emptyset, E_m\subseteq\Omega_{r(k)}\right\}
    \end{align}
\end{subequations}
where $\Omega_{l(k)}$ and $\Omega_{r(k)}$ are the left and right components that form the interface $\Gamma_k$, respectively, with $\Gamma_k = \partial\Omega_{l(k)}\cap\partial\Omega_{r(k)}$.

\paragraph*{Important point:} The critical assumption is that the contact domains $\Omega^{c,-}_r$ and $\Omega^{c,+}_r$ are sufficiently thin such that their transient thermal response is negligible, and thus the contact temperature jumps $\Delta\btheta_r$ are governed by algebraic constraints for heat-flux continuity across the interfaces.
