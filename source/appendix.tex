\subsubsection{Weak Form of Discontinuous Galerkin Method}

Choosing appropriate basis functions $v(x)$ and $w(x)$, the variational formulation of the governing equations is obtained by applying the DG method element-wise to each component of the TPS. In contrast to the standard perfect-contact formulation, the component-to-component interfaces support a finite temperature jump caused by imperfect thermal contact.

Let $\cN_i$ denote the collection of elements neighboring element $E_i$. The following separation is introduced,
\begin{equation}
    \cN_i = \cN_i^{e}\cup\cN_i^{c}
\end{equation}
where $\cN_i^{e}$ denotes the ordinary neighboring elements coupled by the standard DG interface terms through the interface $e_{ij}$, and $\cN_i^{c}$ denotes the neighboring elements separated from $E_i$ by a an imperfect thermal contact interface $\Gamma_{ij}$. Moreover, on the imperfect-contact interface $\Gamma_{ij}$, the interface represents a physical thermal resistance between two neighboring components. Let $R_{x,ij}$ denote the thermal contact resistance and define the corresponding thermal contact conductance as $h_{x,ij} = 1/R_{x,ij}$. Then, the heat flux across the interface is governed by,
\begin{equation}
    q^{(c)}_{ij} = h_{x,ij}\left(T^{-}_i(x,t) - T^{+}_j(x,t)\right), \quad x\in\Gamma_{ij}
\end{equation}
where $T^{-}_i$ and $T^{+}_j$ denote the temperature traces on the minus and plus sides of the interface $\Gamma_{ij}$, respectively. The normal conditions on the interface are given by the conservation of energy across the interface,
\begin{subequations}
    \begin{align}
        -\kappa_i\nabla T^{-}_i(x,t)\cdot\vn_i &= q^{(c)}_{ij}, \quad x\in\Gamma_{ij},\\
        -\kappa_j\nabla T^{+}_j(x,t)\cdot\vn_j &= -q^{(c)}_{ij}, \quad x\in\Gamma_{ij}
    \end{align}
\end{subequations}
where $\vn_i=-\vn_j$ is the unit normal vector of the interface $\Gamma_{ij}$ pointing from component $i$ to component $j$. Thus, across an imperfect contact interface, the heat flux is conserved, while the temperature is allowed to be discontinuous.

Using the Interior Penalty Galerkin (IPG) scheme [Cohen2018], the variational form of Eq. [X] is given by,
\begin{equation}
    \sum_{i=1}^{M}\left(a^e_{\epsilon,i}(v,w) + a^c_{\epsilon,i}(v,w)\right) = \sum_{i=1}^{M} L_i(v)
\end{equation}
where $\epsilon$ is a user-specified parameter, and,
\begin{subequations}
    \begin{align}
        a^e_{\epsilon,i}(v,w) &= \int_{E_i}\left(\rho_i c_{p,i} v\partial_t w + \vk_i\nabla v\cdot\nabla w\right)dE_i - \sum_{j\in\cN_i^{e}\cup\left\{T\right\}}\int_{e_{ij}}\left\{\vk_i\nabla v\cdot \vn_i\right\}\left[w\right]de_{ij}\notag\\
        &\quad + \epsilon \sum_{j\in\cN_i^{e}\cup\left\{T\right\}}\int_{e_{ij}}\left\{\vk_i\nabla w\cdot \vn_i\right\}\left[v\right]de_{ij} + \sigma\sum_{j\in\cN_i^{e}\cup\left\{T\right\}}\int_{e_{ij}}\left[v\right]\left[w\right]de_{ij}\\
        a^c_{\epsilon,i}(v,w) &= -\sum_{l\in\cN_i^{c}}\int_{\Gamma_{il}} h_{x,il}v\left(w^- - w^+\right)d\Gamma_{il}\\
        L_i(v) &= \epsilon\sum_{j\in\cN_i^{e}\cup\left\{T\right\}}\int_{e_{ij}} \left(\vk_i\nabla w\cdot \vn_i\right) T_b de_{ij} + \int_{e_{iq}} v q_b de_{iq} + \sigma\int_{e_{iT}} v T_b de_{iT}
    \end{align}
\end{subequations}
In the bi-linear form above, the notations of $\left[\right]$ and $\left\{\right\}$ are respectively the jumps and averages at a boundary $e_{ij}$ shared by two neighboring elements $E_i$ and $E_j$, defined as,
\begin{equation}
    \left[u\right] = u\Big|_{E_i} - u\Big|_{E_j},\quad \left\{u\right\} = \frac{1}{2}\left(u\Big|_{E_i} + u\Big|_{E_j}\right),\quad x\in e_{ij}=\partial E_i\cap\partial E_j
\end{equation}
Furthermore, the terms associated with $\sigma$ are introduced to enforce the Dirichlet boundary conditions, where $\sigma$ is a user-specified penalty parameter whose value depend on the size of an element. Depending on the choice of $\epsilon$, the bi-linear form may correspond to symmetric IPG ($\epsilon=-1$), non-symmetric IPG ($\epsilon=1$), or incomplete IPG ($\epsilon=0$) schemes. \subsection{Lumped Capacitance Model}

The TPS is divided into $N$ components with $M$ elements. For component $i$, define $\Omega_i$ as the spatial domain of the component, and $\partial\Omega_i$ as the boundary of the component. For each interface between two components, define $l(i)$ and $r(i)$ as the left and right components that form the interface, respectively. 

For interface $i$, define $\Gamma_i=\partial\Omega_{l(i)}\cap\partial\Omega_{r(i)}$,
\begin{subequations}
    \begin{align}
        \theta_i^{-} (t) &= \frac{1}{\left|\Gamma_i\right|}\int_{\Gamma_i} T^{-}_h(x,t)d\Gamma,\\
        \theta_i^{+}(t) &= \frac{1}{\left|\Gamma_i\right|}\int_{\Gamma_i} T^{+}_h(x,t)d\Gamma\\
        \Delta\theta_i(t) &= \theta_i^{-}(t) - \theta_i^{+}(t)
    \end{align}
\end{subequations}
Define the trace operators,
\begin{subequations}
    \begin{align}
        C_i^{-}\vu &= \frac{1}{\left|\Gamma_i\right|}\int_{\Gamma_i} \sum_{j=1}^{N_u} u_j(t)\phi_j(x)d\Gamma,\\
        C_i^{+}\vu &= \frac{1}{\left|\Gamma_i\right|}\int_{\Gamma_i} \sum_{j=1}^{N_u}u_j(t)\phi_j(x)d\Gamma
    \end{align}
\end{subequations}
Stacking all the $N_{\Gamma}$ interfaces, define the trace extraction operator as,
\begin{equation}
    \vPsi^+ = \begin{bmatrix}
        C_1^{-} - C_1^{+}\\
        \vdots\\
        C_{N_{\Gamma}}^{-} - C_{N_{\Gamma}}^{+}
    \end{bmatrix}\in\mathbb{R}^{N_{\Gamma}\times MP}
\end{equation}
where $M$ is the number of elements and $P$ is the number of basis functions per element. Therefore, it follows that,
\begin{equation}
    \vs(t) = \vPsi^+\vu(t)
\end{equation}
Thus,
\begin{equation}
    \vPi^+ = \begin{bmatrix}
        \vPhi^+ \\ 
        \vPsi^+
    \end{bmatrix}
\end{equation}

Introduce the lifting matrix,
\begin{equation}
    \vPsi\in\mathbb{R}^{MP\times N_{\Gamma}}
\end{equation}
such that,
\begin{equation}
    \vPhi^+\vPsi = \vzero,\quad \vPsi^+\vPsi = \vI
\end{equation}

Note that $\Phi\bvu$ contributes $E_{\Gamma}\bvu$ to the interface jump, therefore the state decomposition can be written as,
\begin{equation}
    \vu = \vPhi\bvu + \vPsi\left(\Deltavtheta - E_{\Gamma}\bvu\right) + \delta\vu = \tilde{\vPhi}\bvu + \vPsi\Deltavtheta + \delta\vu
\end{equation}
where $\tilde{\vPhi} = \vPhi - \vPsi E_{\Gamma}$, and $\delta\vu$ is the deviation from the average temperature in the components.
